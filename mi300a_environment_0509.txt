# -- Occupancy and Kernel Sizing ---------------------------------------------
# Max concurrent threads: 228 CUs x 32 waves x 64 threads = 467,072
#
# Minimum blocks for full occupancy:
#   Formula: (32 waves/CU) / (block_threads / 64) x 228 CUs
#   256 threads/block  (4 waves):  8 blocks/CU x 228 =  1,824 blocks
#   512 threads/block  (8 waves):  4 blocks/CU x 228 =    912 blocks
#   1024 threads/block (16 waves): 2 blocks/CU x 228 =    456 blocks
#
# Latency hiding budget:
#   Each SIMD holds 8 wavefronts. HBM load latency ~300 ns = ~630 cycles
#   at 2100 MHz. The 7 non-stalled wavefronts per SIMD fill this gap.
#   Full occupancy (1,824+ blocks at 256 threads) is needed for memory-
#   bandwidth-bound kernels to fully hide HBM latency.
#
# Guidelines:
#   Default:              256 threads/block, 1,824+ blocks  (BW-bound)
#   Compute-bound GEMM:   256 threads/block, 1,824+ blocks
#   Small/latency work:    64 threads/block (1 wavefront per SIMD)
#   Max LDS reuse:        1024 threads/block
#   LDS budget at full occupancy (8 blocks/CU): 64 KB / 8 = 8 KB per block
#   512-1024 total blocks is sufficient for bandwidth-bound kernels.

# 6. P2P INFINITY FABRIC
# ============================================================================

# The correct pattern for concurrent P2P transfers is one dedicated
# hipStream_t per directed link (src, dst pair). Each stream has its own
# HSA command queue (~4096 packets deep); sharing a stream across multiple
# concurrent async copies will exhaust the queue and cause a crash.
#
# Required pattern:
#   hipSetDevice(src);
#   hipStreamCreate(&streams[src][dst]);
#   hipMemcpyPeerAsync(dst_buf, dst, src_buf, src, bytes, streams[src][dst]);
#   hipStreamSynchronize(streams[src][dst]);   // sync after every round
#
# 64-bit atomics on peer memory: unsigned long long* required (same as above).
#
# Enable peer access before any transfer:
#   hipDeviceCanAccessPeer(&can, src, dst);    // check first
#   hipDeviceEnablePeerAccess(dst, 0);         // then enable
#
# Performance on this system:
#   SDMA (hipMemcpyPeerAsync, hipMalloc both ends):  ~90 GB/s per link
#   Kernel remote write (CU load/store over IF):     ~104 GB/s per link
#   Theoretical per-link peak:                       128 GB/s
#   One-way latency (SDMA, small messages):          ~4-8 us
#   Remote atomicAdd (1 target) vs local:            ~2x slower
#
# Environment overrides:
#   HSA_ENABLE_SDMA=0    force CU-driven (blit) copies instead of SDMA
#   HIP_LAUNCH_BLOCKING=1  serialize all kernel launches (debug)
